Programa em C: Roleta

Veja o código abaixo e insira os dados necessários para testar o simulador de roleta.

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

void girarRoleta(char opcoes[][50], int numOpcoes) {
    int sorteado = rand() % numOpcoes;
    printf("A roleta parou em: %s\n", opcoes[sorteado]);
}

int main() {
    int numOpcoes;
    printf("Quantas opções deseja na roleta? ");
    scanf("%d", &numOpcoes);
    
    char opcoes[numOpcoes][50];
    printf("Digite as opções:\n");
    for (int i = 0; i < numOpcoes; i++) {
        scanf("%s", opcoes[i]);
    }
    
    srand(time(NULL));
    girarRoleta(opcoes, numOpcoes);
    
    return 0;
}
